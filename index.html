    # optimize & convert to WebP
curl -X POST https://seraf.ir/api/v1/optimize \
  -H "Authorization: Bearer $SERAF_KEY" \
  -F "image=@photo.jpg" \
  -F "format=webp" -F "quality=80"
    $ch = curl_init('https://seraf.ir/api/v1/optimize');
curl_setopt_array($ch, [
  CURLOPT_POST => true,
  CURLOPT_RETURNTRANSFER => true,
  CURLOPT_HTTPHEADER => ['Authorization: Bearer ' . $key],
  CURLOPT_POSTFIELDS => [
    'image' => new CURLFile('photo.jpg'),
    'response' => 'binary',
  ],
]);
file_put_contents('photo.min.jpg', curl_exec($ch));
